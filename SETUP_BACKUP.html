우리의 하루 · 백업

일정을 파일로 보관하기

1. 새 화면 올리기

  1. 최신 ZIP을 다운로드하고 압축을 풉니다.
  2. GitHub의 본인 저장소 → Code → Add file → Upload files에서 dashboard-v1 폴더 안의 파일 전부를 올립니다. ZIP 자체는 올리지 않습니다.
  3. Commit changes를 누르고 게시가 끝나면 사이트를 새로고침합니다.
  4. 화면에 v1 · 공동 일정가 보이면 오른쪽 위 톱니바퀴를 누릅니다.

2. 보안 규칙 한 번 갱신하기

백업에 담긴 예전 작성자와 시각을 복구할 수 있도록 규칙을 보완했습니다. 가져온 사람과 시각은 별도로 검증·기록하며, 허용된 계정만 사용할 수 있는 제한은 유지합니다. members는 다시 만들지 않습니다.

  1. 아래 보안 규칙 복사를 누릅니다.
  2. Firebase 규칙 화면 열기를 누릅니다. 또는 Firebase → 본인의 Firebase 프로젝트 → Firestore Database → 규칙(Rules)으로 갑니다.
  3. 기존 규칙을 전체 선택해서 지운 뒤 붙여넣고 게시(Publish)합니다.
  4. 대시보드로 돌아와 새로고침합니다.

규칙 내용 펼치기
// 허용 목록 변경은 Firebase 관리 화면에서만 합니다. 앱은 자기 등록 여부만 읽습니다.
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function googleUser() {
      return request.auth != null
        && request.auth.token.email is string
        && request.auth.token.email_verified == true
        && request.auth.token.firebase.sign_in_provider == 'google.com';
    }
    function allowed() {
      return googleUser()
        && get(/databases/$(database)/documents/members/$(request.auth.token.email.lower())).data.enabled == true;
    }
    function actor(value) {
      return value is map && value.keys().hasOnly(['id', 'name', 'email'])
        && value.id == request.auth.uid && value.email == request.auth.token.email
        && value.name is string && value.name.size() > 0 && value.name.size() <= 200;
    }
    // 시작일 없는 기존 문서는 종료일 하루 일정으로 계속 허용합니다.
    function dates(data) {
      let start = data.get('startDate', data.dueDate);
      return (start == null && data.dueDate == null) ||
        (start is string && data.dueDate is string && start <= data.dueDate
        && start.matches('^[0-9]{4}-(0[1-9]|1[0-2])-(0[1-9]|[12][0-9]|3[01])$'));
    }
    // 새 필드가 없는 기존 자료는 종일·공동 일정으로 허용합니다.
    function options(data) {
      let allDay = data.get('allDay', true);
      let start = data.get('startDate', data.dueDate);
      let first = data.get('startTime', null);
      let last = data.get('endTime', null);
      return allDay is bool && data.get('scope', 'shared') in ['shared', 'private']
        && (allDay ? (first == null && last == null) :
          (start is string && data.dueDate is string && first is string && last is string
          && first.matches('^([01][0-9]|2[0-3]):[0-5][0-9]$')
          && last.matches('^([01][0-9]|2[0-3]):[0-5][0-9]$')
          && (start < data.dueDate || (start == data.dueDate && first < last))));
    }
    function content(data) {
      return options(data) && dates(data) && data.title is string && data.title.size() > 0 && data.title.size() <= 120
        && (data.dueDate == null || (data.dueDate is string
          && data.dueDate.matches('^[0-9]{4}-(0[1-9]|1[0-2])-(0[1-9]|[12][0-9]|3[01])$')))
        && data.important is bool && data.completed is bool
        && data.revision is int && data.revision > 0
        && data.updatedAt == request.time && actor(data.updatedBy);
    }
    // 백업 원본의 작성 정보는 사용자 파일의 기록이며, 복구 실행자는 별도로 검증합니다.
    function oldPerson(value) {
      return value is map && value.id is string && value.name is string && value.email is string;
    }
    function oldTime(value) { return value == null || value is timestamp; }
    function restored(data) {
      return options(data) && dates(data) && data.restoredAt == request.time && actor(data.restoredBy)
        && data.title is string && data.title.size() > 0 && data.title.size() <= 120
        && (data.dueDate == null || (data.dueDate is string
          && data.dueDate.matches('^[0-9]{4}-(0[1-9]|1[0-2])-(0[1-9]|[12][0-9]|3[01])$')))
        && data.important is bool && data.completed is bool
        && data.revision is int && data.revision >= 0 && data.schemaVersion == 1
        && oldTime(data.createdAt) && oldTime(data.updatedAt) && oldTime(data.completedAt)
        && (data.completed || data.completedAt == null)
        && oldPerson(data.createdBy) && oldPerson(data.updatedBy);
    }
    // 계정 목록은 공개하지 않습니다. 미등록 사용자도 자기 등록 여부만 확인할 수 있습니다.
    match /members/{email} {
      allow get: if googleUser() && email == request.auth.token.email.lower();
      allow list, write: if false;
    }
    // 색상 설정은 허용된 사람만 읽으며 Firebase 관리 화면에서만 수정합니다.
    match /settings/appearance { allow get: if allowed(); allow list, write: if false; }
    match /tasks/{taskId} {
      allow read: if allowed();
      allow create: if allowed() && content(request.resource.data)
        && request.resource.data.keys().hasOnly(['title', 'dueDate', 'startDate', 'allDay', 'startTime', 'endTime', 'scope', 'important', 'completed',
          'completedAt', 'createdAt', 'createdBy', 'updatedAt', 'updatedBy', 'schemaVersion', 'revision'])
        && request.resource.data.schemaVersion == 1 && request.resource.data.revision == 1
        && request.resource.data.createdAt == request.time && actor(request.resource.data.createdBy)
        && request.resource.data.completedAt == (request.resource.data.completed ? request.time : null);
      allow create: if allowed() && restored(request.resource.data);
      // 작성자·작성 시각 및 나중에 추가한 필드는 그대로 보존합니다.
      allow update: if allowed() && content(request.resource.data)
        && request.resource.data.diff(resource.data).affectedKeys().hasOnly([
          'title', 'dueDate', 'startDate', 'allDay', 'startTime', 'endTime', 'scope', 'important', 'completed', 'completedAt', 'updatedAt', 'updatedBy', 'revision'])
        && request.resource.data.revision == resource.data.get('revision', 0) + 1
        && request.resource.data.completedAt == (request.resource.data.completed
          ? (resource.data.get('completed', false) && resource.data.get('completedAt', null) != null
              ? resource.data.completedAt : request.time) : null);
      allow delete: if allowed();
    }
    // 위에 명시하지 않은 위치(진단·하위 문서 포함)는 모두 차단합니다.
    match /{document=**} { allow read, write: if false; }
  }
}

3. 폰에서 확인하기

  1. 공동 일정에 연결된 상태에서 톱니바퀴 → JSON 내보내기를 누릅니다.
  2. 폰의 내 파일 → 다운로드에서 our-day-backup으로 시작하는 .json 파일을 확인합니다. 완료한 일도 들어 있습니다.
  3. 다시 대시보드 톱니바퀴 → JSON 가져오기 → 방금 파일을 선택합니다.
  4. 추가 0개와 기존 항목 건너뛰기가 표시되는지 확인하고 없는 항목 추가를 누릅니다. 기존 일정이 중복되거나 덮어써지지 않아야 합니다.
  5. 실제 복원까지 시험하려면 중요하지 않은 ‘백업 시험’ 항목을 새로 만들고 다시 내보냅니다. 이 시험 항목만 삭제한 뒤 새 백업을 가져옵니다. 추가 1개가 표시되고 원래 작성 기록과 함께 돌아오면 완료입니다.

삭제는 영구 삭제입니다. 시험 항목 외 실제 일정을 지우지 마세요. 가져오기는 확인 버튼을 누르기 전에는 저장하지 않습니다.

기억할 점

  • 백업 범위는 모든 일정과 완료한 일입니다. 허용 목록(members), 로그인 정보, Firebase 프로젝트 설정은 포함하지 않습니다.
  • 같은 고유번호가 이미 있으면 내용이 달라도 건너뜁니다. 예전 내용으로 되돌리는 덮어쓰기 복원은 제공하지 않습니다.
  • 제목이 같아도 고유번호가 다르면 별개 항목입니다. 다른 프로젝트로 이전할 때도 해당 프로젝트의 로그인·허용 목록·규칙 설정을 먼저 마쳐야 합니다.
  • 가져오기 파일은 10MB 이하입니다. 파일 형식과 모든 항목을 먼저 검사하고, 잘못된 파일은 저장 전에 거절합니다.
  • 진행 중 중단하거나 연결이 끊기면 이미 추가한 항목은 남습니다. 같은 파일을 다시 가져오면 기존 항목을 건너뛰고 나머지를 추가합니다.
  • 작성자·작성/수정/완료 시각은 파일의 원본 기록입니다. 파일 내용의 진위까지 증명하는 것은 아닙니다. 앱은 이번 가져오기 실행자와 시각을 별도로 기록합니다.
  • 백업 파일에는 일정과 이메일이 담깁니다. 공개 GitHub 저장소에 올리지 말고 개인 파일로 보관하세요.
  • 로그아웃·미허용 계정·오프라인 상태에서는 백업 버튼을 사용할 수 없습니다.
